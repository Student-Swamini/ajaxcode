

CREATE TABLE customer (
    cno SERIAL PRIMARY KEY,
    cname VARCHAR(50),
    city VARCHAR(50)
);

CREATE TABLE orders (
    ono SERIAL PRIMARY KEY,
    odate DATE,
    shipping_address VARCHAR(100),
    cno INT REFERENCES customer(cno)
);
